  uniform float uTime;
  uniform float uAspect;
  uniform float uSeed;
  uniform float uVisibility;
  uniform float uLoadingVisibility;
  varying vec2 vUv;

  float hash(float n) {
    return fract(sin(n) * 43758.5453123);
  }

  vec2 hash2(float n) {
    return vec2(hash(n), hash(n + 57.0));
  }

  // Segment SDF from iquilezles.org/articles/distfunctions2d
  float sdSegment(vec2 p, vec2 a, vec2 b) {
    vec2 pa = p - a, ba = b - a;
    float h = clamp(dot(pa, ba) / dot(ba, ba), 0.0, 1.0);
    return length(pa - ba * h);
  }

  void main() {
    vec2 uv = vec2(vUv.x * uAspect, vUv.y);
    
    float maxMargin = 0.35 * 0.5 + 0.1;
    
    vec3 color1 = vec3(0.01);
    vec3 color2 = vec3(0.02);
    
    vec4 color = vec4(0.0);
    
    for (int i = 0; i < 5; i++) {
      float baseSeed = float(i) * 123.456 + uSeed;
      
      // Speed and phase stay fixed per capsule
      float randSpeed = hash(baseSeed + 500.0);
      float randPhase = hash(baseSeed + 600.0);
      
      float speed = 0.3 + randSpeed * 0.4;
      float phase = randPhase * 6.28318;
      float t = uTime * speed + phase;
      
      float fade = sin(t) * 0.5 + 0.5;
      
      // New seed each cycle (when fade = 0)
      float cycle = floor((t + 1.5708) / 6.28318);
      float seed = baseSeed + cycle * 777.0;
      
      vec2 randPos = hash2(seed);
      float randSize = hash(seed + 200.0);
      float randColor = hash(seed + 300.0);
      float randBlur = hash(seed + 400.0);
      
      float segmentHeight = 0.1 + randSize * 0.1;
      float blur = 0.3 + randBlur * 0.4;
      
      float cx = maxMargin + randPos.x * (uAspect - 2.0 * maxMargin);
      float cy = maxMargin + randPos.y * (1.0 - 2.0 * maxMargin);
      
      vec2 a = vec2(cx, cy - segmentHeight * 0.5);
      vec2 b = vec2(cx, cy + segmentHeight * 0.5);
      
      float d = sdSegment(uv, a, b);
      float shape = 1.0 - smoothstep(0.0, blur, d);
      
      vec3 capsuleColor = mix(color1, color2, randColor);
      float intensity = shape * fade;
      color += vec4(capsuleColor * intensity, intensity);
    }
    
    gl_FragColor = vec4(color.rgb, color.a * uVisibility * uLoadingVisibility);

    // TODO: Bunch of state issues with the loader so disabling for now so we can merge this 
    gl_FragColor = vec4(0.0);
  }

attribute vec3 velocity;
attribute float phase;
attribute float sizeScale;

uniform float uSize;
uniform float uSizeVariation;
uniform float uTime;
uniform float uSpeed;
uniform float uTurbulence;
uniform float uBlinkSpeed;
uniform float uBlinkMin;
uniform sampler2D tNoise;

varying float vAlpha;

float sampleNoise(vec2 uv) {
  return texture2D(tNoise, uv).r * 2.0 - 1.0;
}

void main() {
  vec3 animatedPos = position + velocity * uTime * uSpeed * 0.01;
  animatedPos = fract(animatedPos * 0.5 + 0.5) * 2.0 - 1.0;

  if (uTurbulence > 0.0) {
    vec3 noisePos = position * 2.0 + uTime * 0.1;
    animatedPos.x += sampleNoise(noisePos.xy * 0.125) * uTurbulence * 0.1;
    animatedPos.y += sampleNoise(noisePos.yz * 0.125) * uTurbulence * 0.1;
    animatedPos.z += sampleNoise(noisePos.zx * 0.125) * uTurbulence * 0.1;
  }

  vec4 mvPosition = modelViewMatrix * vec4(animatedPos, 1.0);
  
  float variationScale = mix(1.0, sizeScale, uSizeVariation);
  gl_PointSize = uSize * variationScale * (200.0 / -mvPosition.z);

  float blinkWave = sin(uTime * uBlinkSpeed + phase) * 0.5 + 0.5;
  float blink = uBlinkSpeed > 0.0 ? uBlinkMin + (1.0 - uBlinkMin) * blinkWave : 1.0;

  vAlpha = blink;

  gl_Position = projectionMatrix * mvPosition;
}

uniform float uTime;
uniform float uSpeed;
uniform float uScale;

attribute vec4 instanceRandom;

varying vec2 vUv;
varying vec4 vRandom;

void main() {
  vUv = uv;
  vRandom = instanceRandom;
  
  vec3 pos = position;
  
  float flapSpeed = 4.0 + instanceRandom.y * 4.0;
  float flapAmp = 0.8 + instanceRandom.z * 0.5;
  float phase = instanceRandom.x * 6.28;
  
  float t = uTime * uSpeed * flapSpeed + phase;
  float wave = sin(t);
  wave += sin(t * 2.1 + instanceRandom.w * 3.14) * 0.2;
  wave += sin(t * 0.37 + instanceRandom.z * 5.0) * 0.3;
  
  float wingDist = abs(pos.x);
  float wingBlend = smoothstep(0.02, 0.4, wingDist);
  float asymmetry = 1.0 + sign(pos.x) * sin(t * 0.7 + instanceRandom.w * 6.28) * 0.08;
  
  float flapAngle = wave * flapAmp * wingBlend * asymmetry;
  float c = cos(flapAngle);
  float s = sin(flapAngle);
  float newX = pos.x * c - pos.y * s * sign(pos.x);
  float newY = pos.x * s * sign(pos.x) + pos.y * c;
  pos.x = newX;
  pos.y = newY;

  pos.y += smoothstep(1.0, -0.5, wave) * 0.25;
  
  pos *= uScale * (0.8 + instanceRandom.x * 0.4);
  
  vec4 worldPos = modelMatrix * instanceMatrix * vec4(pos, 1.0);
  gl_Position = projectionMatrix * viewMatrix * worldPos;
}

uniform sampler2D tMap;
uniform vec3 uColor;
uniform vec3 uGlowColor;
uniform float uTime;
uniform float uOpacity;
uniform float uHue;

varying vec2 vUv;
varying vec4 vRandom;

vec3 rgb2hsl(vec3 col) {
  float minc = min(col.r, min(col.g, col.b));
  float maxc = max(col.r, max(col.g, col.b));
  vec3 mask = step(col.grr, col.rgb) * step(col.bbg, col.rgb);
  vec3 h = mask * (vec3(0.0, 2.0, 4.0) + (col.gbr - col.brg) / (maxc - minc + 0.000001)) / 6.0;
  return vec3(
    fract(1.0 + h.x + h.y + h.z),
    (maxc - minc) / (1.0 - abs(minc + maxc - 1.0) + 0.000001),
    (minc + maxc) * 0.5
  );
}

vec3 hsl2rgb(vec3 c) {
  vec3 rgb = clamp(abs(mod(c.x * 6.0 + vec3(0.0, 4.0, 2.0), 6.0) - 3.0) - 1.0, 0.0, 1.0);
  return c.z + c.y * (rgb - 0.5) * (1.0 - abs(2.0 * c.z - 1.0));
}

void main() {
  vec2 uv = vUv;
  uv.y += 0.1 * vRandom.x;
  vec4 tex = texture2D(tMap, uv);
  
  vec3 hsl = rgb2hsl(tex.rgb);
  hsl.x = fract(hsl.x + uHue + vRandom.x * 0.05);
  vec3 color = hsl2rgb(hsl) * uColor;
  
  float pulse = sin(uTime * 3.0 + vRandom.a * 6.28) * 0.3 + 0.7;
  float edgeGlow = 2.0 - smoothstep(0.1, 0.9, vUv.x) - smoothstep(0.1, 0.9, vUv.y);
  color += uGlowColor * edgeGlow * 0.3 * pulse;

  color = pow(color, vec3(1.5));
  
  gl_FragColor = vec4(color, tex.a * uOpacity);
}
